# Authorization

## Example Usage

```typescript
import { Authorization } from "@wistia/wistia-api-client/models/operations";

let value: Authorization = {
  type: "account",
  id: "<id>",
  permissions: [
    "show",
    "update",
    "destroy",
    "edit-transcripts",
    "create-folders",
  ],
};
```
