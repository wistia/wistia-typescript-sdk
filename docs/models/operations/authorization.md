  type: "brand",

    "update",
    "destroy",
    "edit-transcripts",
    "view-stats",
    "share",
    "translate",
    "create-folders",
    "create-channels",
    "create-webinars",
    "manage-team",
    "edit",
    "order-audio-descriptions",
    "create-transcripts",
    "view-speakers",
    "view-tags",
    "manage-tags",
    "manage-allowed-domains",
    "upload",
    "manage-deleted-media",
    "create-brands",
    "manage-brand-preload",
  ],
};
